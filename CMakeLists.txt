cmake_minimum_required(VERSION 3.24)
project(PeripheralBattery VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(PBC_OUTPUT_DIR "${CMAKE_SOURCE_DIR}/dist" CACHE PATH "Release executable output")
set(PBC_TEST_DATA_DIR "${CMAKE_SOURCE_DIR}/verification/ui-regression/data" CACHE PATH "Isolated UI test data")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(PeripheralBattery WIN32 src/main.cpp src/hid.cpp src/settings.cpp src/draw.cpp assets/app.rc)
target_compile_definitions(PeripheralBattery PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(PeripheralBattery PRIVATE /utf-8 /W4 /permissive- /EHsc $<$<CONFIG:Release>:/O1>)
target_link_libraries(PeripheralBattery PRIVATE setupapi hid shell32 advapi32 gdiplus dwmapi user32 gdi32 comctl32 psapi imm32)
set_target_properties(PeripheralBattery PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PBC_OUTPUT_DIR}")
target_link_options(PeripheralBattery PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/assets/app.manifest" $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
enable_testing()
add_test(NAME protocol_parsers COMMAND PeripheralBattery --self-test)
add_test(NAME ui_regression COMMAND PeripheralBattery --verify-layout --data-dir "${PBC_TEST_DATA_DIR}")
